cmake_minimum_required(VERSION 3.24)
project(CHUD VERSION 0.1.0 LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
  string(TOUPPER ${config} upper)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${upper} "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_library(chud_frontend compiler/src/lexer.cpp compiler/src/parser.cpp compiler/src/codegen.cpp)
target_include_directories(chud_frontend PUBLIC compiler/include)
add_executable(chudc compiler/src/main.cpp compiler/src/process.cpp)
target_link_libraries(chudc PRIVATE chud_frontend)
target_compile_definitions(chudc PRIVATE CHUD_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(chud tools/chud/launcher.cpp compiler/src/process.cpp)
target_include_directories(chud PRIVATE compiler/include)
target_compile_definitions(chud PRIVATE CHUD_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(chud-lsp tools/chud/launcher.cpp compiler/src/process.cpp)
target_include_directories(chud-lsp PRIVATE compiler/include)
target_compile_definitions(chud-lsp PRIVATE CHUD_ROOT="${CMAKE_CURRENT_SOURCE_DIR}" CHUD_TOOL_SCRIPT="/tools/chud-lsp/server.mjs")
if(MSVC)
  target_compile_options(chud_frontend PRIVATE /W4 /utf-8)
  target_compile_options(chudc PRIVATE /W4 /utf-8)
endif()
enable_testing()
find_program(NODE_EXECUTABLE node REQUIRED)
add_test(NAME compiler COMMAND ${NODE_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/compiler.test.mjs)
set_tests_properties(compiler PROPERTIES ENVIRONMENT "CHUD_COMPILER=$<TARGET_FILE:chudc>")
